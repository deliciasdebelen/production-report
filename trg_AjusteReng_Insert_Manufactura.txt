
CREATE TRIGGER [dbo].[trg_AjusteReng_Insert_Manufactura]
ON [dbo].[saAjusteReng]
AFTER INSERT
AS
BEGIN
    SET NOCOUNT ON;

    -- Paso 1: Ajustes con 'MANUFACTURA' y extracción de odp_num desde el motivo
    DECLARE @AjustesManufactura TABLE (Ajue_num INT, odp_num INT);
    
    INSERT INTO @AjustesManufactura (Ajue_num, odp_num)
    SELECT DISTINCT 
        i.Ajue_num,
        TRY_CAST(
            LTRIM(RTRIM(
                SUBSTRING(
                    a.motivo,
                    CHARINDEX('ODP:', a.motivo) + 4,
                    CASE 
                        WHEN CHARINDEX(' ', a.motivo, CHARINDEX('ODP:', a.motivo) + 4) > 0 
                        THEN CHARINDEX(' ', a.motivo, CHARINDEX('ODP:', a.motivo) + 4) - (CHARINDEX('ODP:', a.motivo) + 4)
                        ELSE LEN(a.motivo) - (CHARINDEX('ODP:', a.motivo) + 3)
                    END
                )
            )) AS INT
        ) AS odp_num
    FROM inserted i
    INNER JOIN saAjuste a ON i.Ajue_num = a.Ajue_num
    WHERE a.motivo LIKE '%MANUFACTURA%'
      AND a.motivo LIKE '%ODP:%';

    -- Eliminar registros donde no se pudo extraer odp_num
    DELETE FROM @AjustesManufactura WHERE odp_num IS NULL;
    IF NOT EXISTS (SELECT 1 FROM @AjustesManufactura)
        RETURN;

    -- Paso 2: Renglones E-PTD recién insertados
    DECLARE @EPTD_Insertados TABLE (Ajue_num INT, Reng_num INT);
    INSERT INTO @EPTD_Insertados (Ajue_num, Reng_num)
    SELECT i.Ajue_num, i.Reng_num
    FROM inserted i
    INNER JOIN @AjustesManufactura am ON i.Ajue_num = am.Ajue_num
    WHERE i.co_tipo = 'E-PTD';

    IF NOT EXISTS (SELECT 1 FROM @EPTD_Insertados)
        RETURN;

    -- Paso 3: Calcular costo total de insumos y obtener unidades producidas desde NSPOrdenproduccion
    DECLARE @Calculos TABLE (
        Ajue_num INT,
        CostoTotalInsumos DECIMAL(18, 6),
        UnidadesProducidas DECIMAL(18, 6)
    );

    INSERT INTO @Calculos (Ajue_num, CostoTotalInsumos, UnidadesProducidas)
    SELECT
        am.Ajue_num,
        -- Costo total de insumos (solo no E-PTD)
        ISNULL((
            SELECT SUM(r.total_art * r.cost_unit)
            FROM saAjusteReng r
            WHERE r.Ajue_num = am.Ajue_num
              AND r.co_tipo <> 'E-PTD'
              AND r.total_art IS NOT NULL
              AND r.cost_unit IS NOT NULL
        ), 0) AS CostoTotalInsumos,
        -- Unidades producidas: tomadas de NSPOrdenproduccion.cantidad usando odp_num
        ISNULL(o.cantidad, 0) AS UnidadesProducidas
    FROM @AjustesManufactura am
    LEFT JOIN [CARMAL_M].[dbo].[NSPOrdenproduccion] o ON am.odp_num = o.odp_num;

    -- Paso 4: Calcular costo unitario final
    DECLARE @CostoAsignar TABLE (Ajue_num INT, CostoUnitarioEPTD DECIMAL(18, 6));
    INSERT INTO @CostoAsignar (Ajue_num, CostoUnitarioEPTD)
    SELECT
        Ajue_num,
        CASE
            WHEN UnidadesProducidas = 0 THEN 0
            ELSE CostoTotalInsumos / UnidadesProducidas
        END AS CostoUnitarioEPTD
    FROM @Calculos;

    -- Paso 5: Actualizar los renglones E-PTD insertados
    UPDATE r
    SET cost_unit = ca.CostoUnitarioEPTD
    FROM saAjusteReng r
    INNER JOIN @EPTD_Insertados e ON r.Ajue_num = e.Ajue_num AND r.Reng_num = e.Reng_num
    INNER JOIN @CostoAsignar ca ON r.Ajue_num = ca.Ajue_num
    WHERE r.co_tipo = 'E-PTD';
END;
